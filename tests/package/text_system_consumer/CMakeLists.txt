cmake_minimum_required(VERSION 3.16)
project(TextSystemConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
add_executable(text_system_consumer ../../unit/text_system_tests.cpp)
target_link_libraries(text_system_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_features(text_system_consumer PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(text_system_consumer PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(text_system_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed_text_system_contract COMMAND text_system_consumer)
