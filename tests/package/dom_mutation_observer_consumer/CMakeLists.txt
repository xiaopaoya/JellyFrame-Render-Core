cmake_minimum_required(VERSION 3.20)
project(DomMutationObserverConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_SOURCE_HASH STREQUAL EXPECTED_SOURCE_HASH)
    message(FATAL_ERROR "Installed source identity mismatch")
endif()
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
    message(FATAL_ERROR "Installed engine ABI mismatch")
endif()
file(READ "${JellyFrameRenderCore_PROFILE_FILE}" profile_json)
string(JSON installed_abi GET "${profile_json}" engineAbi)
if(NOT installed_abi STREQUAL JellyFrameRenderCore_ENGINE_ABI)
    message(FATAL_ERROR "Profile and package ABI disagree")
endif()
file(READ "${JellyFrameRenderCore_SOURCE_MANIFEST_FILE}" manifest_json)
string(JSON manifest_hash GET "${manifest_json}" sourceHash)
if(NOT manifest_hash STREQUAL EXPECTED_SOURCE_HASH)
    message(FATAL_ERROR "Manifest and expected source identity disagree")
endif()
string(JSON installed_profile GET "${profile_json}" profileId)
if(NOT installed_profile STREQUAL EXPECTED_PROFILE_ID)
    message(FATAL_ERROR "Installed feature profile mismatch")
endif()
add_executable(dom_mutation_observer_consumer ../../unit/dom_mutation_observer_tests.cpp)
target_link_libraries(dom_mutation_observer_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_features(dom_mutation_observer_consumer PRIVATE cxx_std_17)
enable_testing()
add_test(NAME installed_dom_mutation_observer_contract COMMAND dom_mutation_observer_consumer)
