cmake_minimum_required(VERSION 3.16)
project(EventDispatchCompletionConsumer LANGUAGES CXX)

find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT DEFINED EXPECTED_CORE_SOURCE_HASH OR NOT JellyFrameRenderCore_SOURCE_HASH STREQUAL EXPECTED_CORE_SOURCE_HASH)
    message(FATAL_ERROR "Expected installed source hash is required and must match")
endif()
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
    message(FATAL_ERROR "Event completion consumer requires Core ABI4")
endif()
file(READ "${JellyFrameRenderCore_PROFILE_FILE}" profile_json)
string(JSON profile_id GET "${profile_json}" profileId)
if(NOT DEFINED EXPECTED_PROFILE_ID OR NOT profile_id STREQUAL EXPECTED_PROFILE_ID)
    message(FATAL_ERROR "Expected installed feature profile is required and must match")
endif()

add_executable(event_dispatch_completion_consumer main.cpp)
target_compile_features(event_dispatch_completion_consumer PRIVATE cxx_std_17)
target_link_libraries(event_dispatch_completion_consumer PRIVATE JellyFrame::jellyframe_render_core)
enable_testing()
add_test(NAME installed_event_dispatch_completion_contract COMMAND event_dispatch_completion_consumer)
