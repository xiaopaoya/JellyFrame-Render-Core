cmake_minimum_required(VERSION 3.20)
project(BackgroundMultilayerInstalledConsumer LANGUAGES CXX)

find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
    message(FATAL_ERROR "F2a consumer requires ABI4")
endif()
foreach(identity IN ITEMS SOURCE_HASH)
    if(NOT DEFINED EXPECTED_CORE_${identity} OR "${EXPECTED_CORE_${identity}}" STREQUAL "")
        message(FATAL_ERROR "Set EXPECTED_CORE_${identity} to the qualified package identity")
    endif()
    if(NOT "${JellyFrameRenderCore_${identity}}" STREQUAL "${EXPECTED_CORE_${identity}}")
        message(FATAL_ERROR "F2a consumer refuses Core ${identity} mismatch")
    endif()
endforeach()
if(NOT DEFINED EXPECTED_CORE_PROFILE_ID OR "${EXPECTED_CORE_PROFILE_ID}" STREQUAL "")
    message(FATAL_ERROR "Set EXPECTED_CORE_PROFILE_ID to the qualified package profile")
endif()
if(NOT JellyFrameRenderCore_PROFILE_ID STREQUAL EXPECTED_CORE_PROFILE_ID)
    message(FATAL_ERROR "F2a consumer refuses Core PROFILE_ID mismatch")
endif()
file(READ "${JellyFrameRenderCore_PROFILE_FILE}" installed_profile)
string(JSON installed_profile_abi GET "${installed_profile}" engineAbi)
string(JSON installed_profile_id GET "${installed_profile}" profileId)
if(NOT installed_profile_abi STREQUAL "4" OR
   NOT installed_profile_id STREQUAL EXPECTED_CORE_PROFILE_ID)
    message(FATAL_ERROR "F2a consumer refuses installed profile metadata mismatch")
endif()
file(READ "${JellyFrameRenderCore_SOURCE_MANIFEST_FILE}" installed_manifest)
string(JSON installed_source_hash GET "${installed_manifest}" sourceHash)
if(NOT installed_source_hash STREQUAL EXPECTED_CORE_SOURCE_HASH)
    message(FATAL_ERROR "F2a consumer refuses installed source manifest mismatch")
endif()

add_executable(background_multilayer_consumer ../../unit/background_multilayer_style_tests.cpp)
target_link_libraries(background_multilayer_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_definitions(background_multilayer_consumer PRIVATE F2A_INSTALLED_CONSUMER=1)
target_compile_features(background_multilayer_consumer PRIVATE cxx_std_17)

add_executable(background_multilayer_header_only header_only.cpp)
target_include_directories(background_multilayer_header_only PRIVATE
    "$<TARGET_PROPERTY:JellyFrame::jellyframe_render_core,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_features(background_multilayer_header_only PRIVATE cxx_std_17)

enable_testing()
add_test(NAME background_multilayer_installed COMMAND background_multilayer_consumer)
add_test(NAME background_multilayer_header_only COMMAND background_multilayer_header_only)
