cmake_minimum_required(VERSION 3.20)
project(ImageCallbackConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT DEFINED EXPECTED_CORE_SOURCE_HASH OR NOT JellyFrameRenderCore_SOURCE_HASH STREQUAL EXPECTED_CORE_SOURCE_HASH)
    message(FATAL_ERROR "Expected installed source hash is required and must match")
endif()
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
    message(FATAL_ERROR "Image callback consumer requires ABI 4")
endif()
file(READ "${JellyFrameRenderCore_PROFILE_FILE}" installed_profile)
string(JSON installed_profile_id GET "${installed_profile}" profileId)
if(NOT DEFINED EXPECTED_PROFILE_ID OR NOT installed_profile_id STREQUAL EXPECTED_PROFILE_ID)
    message(FATAL_ERROR "Expected installed feature profile is required and must match")
endif()
add_executable(image_callback_consumer ../../unit/image_callback_contract_tests.cpp)
target_compile_features(image_callback_consumer PRIVATE cxx_std_17)
target_link_libraries(image_callback_consumer PRIVATE JellyFrame::jellyframe_render_core)
enable_testing()
add_test(NAME installed_image_callback_contract COMMAND image_callback_consumer)
