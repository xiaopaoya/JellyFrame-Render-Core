cmake_minimum_required(VERSION 3.20)
project(NaturalImageConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT DEFINED EXPECTED_CORE_SOURCE_HASH OR NOT JellyFrameRenderCore_SOURCE_HASH STREQUAL EXPECTED_CORE_SOURCE_HASH)
    message(FATAL_ERROR "Expected installed source hash is required and must match")
endif()
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
    message(FATAL_ERROR "Natural image layout consumer requires ABI 4")
endif()
add_executable(natural_image_consumer main.cpp)
target_compile_features(natural_image_consumer PRIVATE cxx_std_17)
target_link_libraries(natural_image_consumer PRIVATE JellyFrame::jellyframe_render_core)
enable_testing()
add_test(NAME installed_natural_image_contract COMMAND natural_image_consumer)
