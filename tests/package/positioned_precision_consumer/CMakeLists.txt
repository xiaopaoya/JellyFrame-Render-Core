cmake_minimum_required(VERSION 3.20)
project(C5PositionedInstalledConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
  message(FATAL_ERROR "This consumer requires Core ABI4")
endif()
if(NOT DEFINED EXPECTED_SOURCE_HASH OR NOT DEFINED EXPECTED_PROFILE)
  message(FATAL_ERROR "Pass the exact source hash and profile")
endif()
if(NOT JellyFrameRenderCore_SOURCE_HASH STREQUAL EXPECTED_SOURCE_HASH OR
   NOT JellyFrameRenderCore_PROFILE_ID STREQUAL EXPECTED_PROFILE)
  message(FATAL_ERROR "Installed source/profile differs from the requested candidate")
endif()
file(READ "${JellyFrameRenderCore_SOURCE_MANIFEST_FILE}" source_manifest)
string(JSON manifest_hash GET "${source_manifest}" sourceHash)
string(JSON manifest_count GET "${source_manifest}" sourceFileCount)
if(NOT manifest_hash STREQUAL EXPECTED_SOURCE_HASH OR
   NOT manifest_count STREQUAL JellyFrameRenderCore_SOURCE_FILE_COUNT)
  message(FATAL_ERROR "Installed source manifest disagrees with package exports")
endif()
message(STATUS "Qualified source ${manifest_hash}/${manifest_count}; profile ${EXPECTED_PROFILE}")
add_executable(c5_positioned_consumer main.cpp)
target_compile_features(c5_positioned_consumer PRIVATE cxx_std_17)
target_link_libraries(c5_positioned_consumer PRIVATE JellyFrame::jellyframe_render_core)
enable_testing()
add_test(NAME c5_positioned_consumer COMMAND c5_positioned_consumer)
set_tests_properties(c5_positioned_consumer PROPERTIES TIMEOUT 30)
