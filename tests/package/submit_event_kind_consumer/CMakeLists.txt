cmake_minimum_required(VERSION 3.20)
project(SubmitEventKindInstalledConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
  message(FATAL_ERROR "This consumer requires Core ABI4")
endif()
foreach(identity IN ITEMS SOURCE_HASH PROFILE_ID)
  if(NOT DEFINED EXPECTED_${identity} OR "${EXPECTED_${identity}}" STREQUAL "" OR
     NOT "${JellyFrameRenderCore_${identity}}" STREQUAL "${EXPECTED_${identity}}")
    message(FATAL_ERROR "Installed Core ${identity} does not match the requested candidate")
  endif()
endforeach()
add_executable(submit_event_kind_consumer main.cpp)
target_compile_features(submit_event_kind_consumer PRIVATE cxx_std_17)
target_link_libraries(submit_event_kind_consumer PRIVATE JellyFrame::jellyframe_render_core)
enable_testing()
add_test(NAME installed_submit_event_kind_contract COMMAND submit_event_kind_consumer)
set_tests_properties(installed_submit_event_kind_contract PROPERTIES TIMEOUT 30)
