cmake_minimum_required(VERSION 3.16)
project(CanonicalFontResourceConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
add_executable(font_resource_consumer ../../unit/font_resource_tests.cpp)
target_link_libraries(font_resource_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_features(font_resource_consumer PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(font_resource_consumer PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(font_resource_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed_font_resource_contract COMMAND font_resource_consumer)
