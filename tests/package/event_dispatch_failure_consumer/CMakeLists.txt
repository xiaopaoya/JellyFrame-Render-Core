cmake_minimum_required(VERSION 3.20)
project(EventDispatchFailureInstalledConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4" OR
   NOT JellyFrameRenderCore_PACKAGE_VERSION STREQUAL "0.9.0-dev")
  message(FATAL_ERROR "This consumer requires the development ABI4 package")
endif()
foreach(identity IN ITEMS SOURCE_HASH PROFILE_ID)
  if(NOT DEFINED EXPECTED_${identity} OR "${EXPECTED_${identity}}" STREQUAL "" OR
     NOT "${JellyFrameRenderCore_${identity}}" STREQUAL "${EXPECTED_${identity}}")
    message(FATAL_ERROR "Installed Core ${identity} does not match the requested candidate")
  endif()
endforeach()
file(READ "${JellyFrameRenderCore_SOURCE_MANIFEST_FILE}" source_manifest)
string(JSON manifest_hash GET "${source_manifest}" sourceHash)
string(JSON manifest_count GET "${source_manifest}" sourceFileCount)
if(NOT manifest_hash STREQUAL EXPECTED_SOURCE_HASH OR
   NOT manifest_count STREQUAL JellyFrameRenderCore_SOURCE_FILE_COUNT)
  message(FATAL_ERROR "Installed source manifest disagrees with package exports")
endif()
add_executable(event_dispatch_failure_consumer main.cpp)
target_link_libraries(event_dispatch_failure_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_features(event_dispatch_failure_consumer PRIVATE cxx_std_17)
enable_testing()
add_test(NAME event_dispatch_failure_installed COMMAND event_dispatch_failure_consumer)
set_tests_properties(event_dispatch_failure_installed PROPERTIES TIMEOUT 30)
