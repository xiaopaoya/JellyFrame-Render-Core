cmake_minimum_required(VERSION 3.16)
project(FlexPrecisionConsumer LANGUAGES CXX)
find_package(JellyFrameRenderCore CONFIG REQUIRED)
if(NOT JellyFrameRenderCore_ENGINE_ABI STREQUAL "4")
  message(FATAL_ERROR "Expected development ABI 4")
endif()
add_executable(flex_precision_consumer main.cpp)
target_link_libraries(flex_precision_consumer PRIVATE JellyFrame::jellyframe_render_core)
target_compile_features(flex_precision_consumer PRIVATE cxx_std_17)
